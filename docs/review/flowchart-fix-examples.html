    case Mode::eREAD:  g = 1; break;

A continue in a switch case

SWE3-V4-11 · Return not joined to End

int count;
void countItems(const int* items, int n) {
    for (int i = 0; i < n; i++) {
        switch (items[i]) {
        case 0:
            continue;
        case 1:
            count += 2;
            break;
        default:
            break;
        }
        count++;
    }
}
